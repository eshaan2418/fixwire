cmake_minimum_required(VERSION 3.25)
project(fixwire LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(FIXWIRE_SANITIZE "" CACHE STRING "address,undefined | thread | empty")
option(FIXWIRE_FUZZ "Build the libFuzzer parser target" OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(FIXWIRE_SANITIZE)
  add_compile_options(-fsanitize=${FIXWIRE_SANITIZE} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${FIXWIRE_SANITIZE})
endif()
add_library(fixwire INTERFACE)
target_include_directories(fixwire INTERFACE include)
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE SYSTEM)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
foreach(t gtest gtest_main gmock gmock_main)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE -w)
  endif()
endforeach()
enable_testing()
file(GLOB TEST_SOURCES tests/test_*.cpp)
add_executable(fixwire-tests ${TEST_SOURCES})
target_link_libraries(fixwire-tests PRIVATE fixwire GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(fixwire-tests DISCOVERY_TIMEOUT 120)
foreach(tool fixwire-bench fixwire-exchange fixwire-client)
  string(REPLACE "fixwire-" "" src ${tool})
  add_executable(${tool} tools/${src}.cpp)
  target_link_libraries(${tool} PRIVATE fixwire)
endforeach()
if(FIXWIRE_FUZZ)
  add_executable(fuzz-parser fuzz/fuzz_parser.cpp)
  target_compile_options(fuzz-parser PRIVATE -fsanitize=fuzzer,address,undefined -g)
  target_link_options(fuzz-parser PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(fuzz-parser PRIVATE fixwire)
endif()
